### 2. bsp_board.h 核心内容

```c
#ifndef BSP_BOARD_H
#define BSP_BOARD_H

#include <stdint.h>
#include <stdbool.h>

#ifdef __cplusplus
extern "C" {
#endif

/* ================================================================
 * 统一状态码
 * ================================================================ */

typedef enum
{
    BSP_OK = 0,         /* 成功 */
    BSP_ERROR,          /* 通用错误 */
    BSP_BUSY,           /* 忙状态 */
    BSP_ETIMEOUT,       /* 超时 */
    BSP_EINVAL,         /* 参数错误 */
    BSP_ENOMEM,         /* 内存不足 */
    BSP_ENODEV,         /* 设备不存在 */
    BSP_EIO,            /* I/O 错误 */
} bsp_status_t;

/* ================================================================
 * 常用宏
 * ================================================================ */

/* 超时常量：封装 HAL 依赖 */
#define BSP_WAIT_FOREVER  0xFFFFFFFFU  /* 替代 HAL_MAX_DELAY */
#define BSP_NO_WAIT       0U           /* 立即返回 */

/* 数值宏加括号 */
#define BSP_ARRAY_SIZE(arr)  (sizeof(arr) / sizeof((arr)[0]))

/* ================================================================
 * 断言宏
 * ================================================================ */

#ifdef BSP_DEBUG
    #define BSP_ASSERT(expr)                                    \
        do {                                                     \
            if (!(expr)) {                                       \
                return BSP_EINVAL;                               \
            }                                                    \
        } while(0)
#else
    #define BSP_ASSERT(expr) ((void)0)
#endif

/* ================================================================
 * 状态码字符串转换
 * ================================================================ */

const char *bsp_status_str(bsp_status_t status);

#ifdef __cplusplus
}
#endif

#endif /* BSP_BOARD_H */
```

### 3. port_critical.h 核心内容

```c
#ifndef PORT_CRITICAL_H
#define PORT_CRITICAL_H

#include <stdint.h>
#include "bsp_board.h"

#ifdef __cplusplus
extern "C" {
#endif

/* ================================================================
 * 临界区接口
 * ================================================================ */

/**
 * @brief 进入临界区
 * @note 支持嵌套调用，裸机用中断开关实现
 */
void port_enter_critical(void);

/**
 * @brief 退出临界区
 * @note 支持嵌套调用
 */
void port_exit_critical(void);

#ifdef __cplusplus
}
#endif

#endif /* PORT_CRITICAL_H */
```

## 验收标准

1. **编译通过**：项目能空编译通过，无任何业务代码
2. **目录结构**：所有目录和文件创建完成
3. **头文件规范**：所有头文件有 `extern "C"` 守卫
4. **状态码完整**：`bsp_status_t` 枚举包含所有必要状态码
5. **断言宏可用**：`BSP_ASSERT` 宏定义正确
